cmake_minimum_required(VERSION 3.20)
if(WIN32)
    set(HOST_LANGUAGES RC)
    if(NOT CMAKE_C_COMPILER AND CMAKE_CXX_COMPILER)         # SDL3 is C: the same compiler as the rest
        string(REPLACE "clang++" "clang" CMAKE_C_COMPILER "${CMAKE_CXX_COMPILER}")
    endif()
    if(NOT CMAKE_RC_COMPILER AND CMAKE_CXX_COMPILER MATCHES "clang[+][+]")    # the resources (speed.rc): LLVM's own rc beside it
        get_filename_component(LLVM_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
        set(CMAKE_RC_COMPILER "${LLVM_BIN}/llvm-rc.exe")
    endif()
    set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)   # no Visual C++ runtime needed on the player's machine
endif()
project(hp2recomp C CXX ${HOST_LANGUAGES})

# Generated by tools/recomp/gen.py
set(GEN_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../work/recomp/gen)
file(GLOB GEN_SOURCES CONFIGURE_DEPENDS ${GEN_DIR}/*.cpp)

# The renderer: Vulkan, with the headers and glslc of the Vulkan SDK (VULKAN_SDK); the loader itself is opened at run time.
file(GLOB VULKAN_SDK_GUESS C:/VulkanSDK/*)
find_path(VULKAN_INCLUDE vulkan/vulkan.h HINTS $ENV{VULKAN_SDK}/Include ${VULKAN_SDK_GUESS} PATH_SUFFIXES Include)
find_program(GLSLC glslc HINTS $ENV{VULKAN_SDK}/Bin ${VULKAN_SDK_GUESS} PATH_SUFFIXES Bin)
if(NOT VULKAN_INCLUDE OR NOT GLSLC)
    message(FATAL_ERROR "The Vulkan SDK is needed: its headers and glslc")
endif()
set(SHADER_DIR ${CMAKE_CURRENT_SOURCE_DIR}/runtime/shaders)
set(SHADER_OUT ${CMAKE_CURRENT_BINARY_DIR}/shaders)
set(SHADER_HEADERS)
foreach(s native.vert native.frag native_fold.vert native_fold.frag native_post.vert native_post.frag native_restore.frag native_grade.frag native_opaque.frag)
    add_custom_command(OUTPUT ${SHADER_OUT}/${s}.spv.h
        COMMAND ${CMAKE_COMMAND} -E make_directory ${SHADER_OUT}
        COMMAND ${GLSLC} --target-env=vulkan1.1 -O -mfmt=c -o ${SHADER_OUT}/${s}.spv.h ${SHADER_DIR}/${s}
        DEPENDS ${SHADER_DIR}/${s}
        COMMENT "glslc ${s}")
    list(APPEND SHADER_HEADERS ${SHADER_OUT}/${s}.spv.h)
endforeach()
# the scene's shaders again with the far plane cut by a clip distance, not a discard: what keeps early depth for the draws without an alpha test
foreach(s native.vert native.frag)
    add_custom_command(OUTPUT ${SHADER_OUT}/clip_${s}.spv.h
        COMMAND ${CMAKE_COMMAND} -E make_directory ${SHADER_OUT}
        COMMAND ${GLSLC} --target-env=vulkan1.1 -O -DCLIP_FAR -mfmt=c -o ${SHADER_OUT}/clip_${s}.spv.h ${SHADER_DIR}/${s}
        DEPENDS ${SHADER_DIR}/${s}
        COMMENT "glslc ${s} CLIP_FAR")
    list(APPEND SHADER_HEADERS ${SHADER_OUT}/clip_${s}.spv.h)
endforeach()
add_custom_target(native_shaders DEPENDS ${SHADER_HEADERS})

# Sound: STREAM.IRX on the IOP interpreter, libsd on the software SPU2, host output
set(SOUND_SOURCES runtime/iop_cpu.cpp runtime/iop_kernel.cpp runtime/iop_libsd.cpp runtime/spu2.cpp runtime/audio_wasapi.cpp)

# The game's program is not in the exe: data/SLUS_203.62, from the player's disc (setup.cpp, install.cpp)
set(HOST_SOURCES runtime/host.cpp runtime/disc.cpp)
if(WIN32)
    list(APPEND HOST_SOURCES runtime/speed.rc)
    set_source_files_properties(runtime/speed.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/runtime/speed.ico;${CMAKE_CURRENT_SOURCE_DIR}/data/zdir_names_ntsc.txt;${CMAKE_CURRENT_SOURCE_DIR}/data/README.txt;${CMAKE_CURRENT_SOURCE_DIR}/data/ui/cursor.cur;${CMAKE_CURRENT_SOURCE_DIR}/data/lang/ENGLISH.txt;${CMAKE_CURRENT_SOURCE_DIR}/data/lang/FRENCH.txt;${CMAKE_CURRENT_SOURCE_DIR}/data/lang/GERMAN.txt;${CMAKE_CURRENT_SOURCE_DIR}/data/lang/ITALIAN.txt;${CMAKE_CURRENT_SOURCE_DIR}/data/lang/SPANISH.txt;${CMAKE_CURRENT_SOURCE_DIR}/data/lang/SWEDISH.txt")
endif()

# Gamepads: SDL3 inside the exe, nothing of it but the gamepad subsystem
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
foreach(sub AUDIO VIDEO RENDER GPU CAMERA DIALOG TRAY POWER SENSOR)
    set(SDL_${sub} OFF CACHE BOOL "" FORCE)
endforeach()
# its source: the unpacked one of the working tree, else third_party/SDL3-3.4.18.tar.gz unpacked here
set(SDL3_SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/../work/ref/sdl3/SDL3-3.4.18)
if(NOT EXISTS ${SDL3_SOURCE}/CMakeLists.txt)
    set(SDL3_SOURCE ${CMAKE_CURRENT_BINARY_DIR}/sdl3-src/SDL3-3.4.18)
    if(NOT EXISTS ${SDL3_SOURCE}/CMakeLists.txt)
        file(ARCHIVE_EXTRACT INPUT ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/SDL3-3.4.18.tar.gz DESTINATION ${CMAKE_CURRENT_BINARY_DIR}/sdl3-src)
    endif()
endif()
add_subdirectory(${SDL3_SOURCE} ${CMAKE_CURRENT_BINARY_DIR}/sdl3 EXCLUDE_FROM_ALL)

add_executable(hp2recomp runtime/ps2.cpp runtime/pcmenu.cpp runtime/pcsettings.cpp runtime/pclang.cpp runtime/setup.cpp runtime/install.cpp runtime/pcplatform.cpp runtime/pad.cpp runtime/native.cpp runtime/native_tex.cpp runtime/native_vk.cpp runtime/hle_iop.cpp runtime/dma.cpp runtime/vif.cpp runtime/profile.cpp runtime/hot_math.cpp runtime/native_mem.cpp runtime/native_movie.cpp runtime/gs_regs.cpp runtime/pcconfig.cpp runtime/window.cpp ${SOUND_SOURCES} ${HOST_SOURCES} ${GEN_SOURCES})
target_include_directories(hp2recomp PRIVATE runtime ${GEN_DIR} ${VULKAN_INCLUDE} ${SHADER_OUT})
# the playable build is recompwin/speed.exe, next to data/, hp2.ini and saves/
set_target_properties(hp2recomp PROPERTIES CXX_STANDARD 17 OUTPUT_NAME speed RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/../recompwin)
if(WIN32)
    set_target_properties(hp2recomp PROPERTIES WIN32_EXECUTABLE ON)     # no console
    target_link_options(hp2recomp PRIVATE -Wl,/entry:mainCRTStartup)    # still main()
    # the common controls of today for the setup window (setup.cpp); the game's own window has none
    target_link_options(hp2recomp PRIVATE "LINKER:/manifestdependency:type='win32' name='Microsoft.Windows.Common-Controls' version='6.0.0.0' processorArchitecture='*' publicKeyToken='6595b64144ccf1df' language='*'")
endif()
add_dependencies(hp2recomp native_shaders)
# -O1 keeps the 20+ MB of generated code compiling in reasonable time
target_link_options(hp2recomp PRIVATE "-Wl,/map:${CMAKE_BINARY_DIR}/speed.map")   # names for tools/recomp/profrip.py
target_compile_options(hp2recomp PRIVATE -O2 -w -fno-strict-aliasing)
set_source_files_properties(runtime/iop_cpu.cpp runtime/spu2.cpp runtime/native.cpp runtime/native_tex.cpp PROPERTIES COMPILE_OPTIONS "-O2")
set_source_files_properties(runtime/hot_math.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")     # no fused multiply-add: its rounding is not the game's
set_source_files_properties(runtime/native_movie.cpp PROPERTIES COMPILE_OPTIONS "-fwrapv")   # the game's arithmetic wraps
target_link_libraries(hp2recomp PRIVATE user32 gdi32 ole32 comctl32 comdlg32 shell32 uuid SDL3::SDL3-static)
